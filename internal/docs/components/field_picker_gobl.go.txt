import "github.com/invopop/popui.go/goblfields"

// Built once and reused: a tree only changes when the gobl dependency does.
var invoiceFields = goblfields.MustBuild("bill/invoice")

// Or, with the depth capped and every error handled:
fields, err := goblfields.Build("bill/invoice", goblfields.Options{MaxDepth: 3})
if err != nil {
    return err
}
